# Remove everything (containers, volumes, images)
docker compose down --volumes --remove-orphans --rmi all

# Build from scratch
docker compose build --no-cache

# Start containers (detached)
docker compose up -d

# Start with build and view logs
docker compose up --build

#For DB
docker compose exec naqel_maps python manage.py makemigrations
docker compose exec naqel_maps python manage.py migrate

# Tests
docker compose exec naqel_maps python manage.py test mapping layer_uploader
