# Remove everything (containers, volumes, images)
docker compose down --volumes --remove-orphans --rmi all

# Build from scratch
docker compose build --no-cache

# Start containers (detached)
docker compose up -d

# Start with build and view logs
docker compose up --build

# DB
docker compose exec geotrak_maps python manage.py makemigrations
docker compose exec geotrak_maps python manage.py migrate

# Tests
docker compose exec geotrak_maps python manage.py test notifications mapping layer_uploader system_admin --keepdb --noinput
