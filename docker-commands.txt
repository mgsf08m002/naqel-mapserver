================================================================================
                    DOCKER COMMANDS REFERENCE GUIDE
                         For GIS Project
================================================================================

PROJECT SERVICES:
  - db (PostgreSQL):        Container: naqel_maps_db, Port: 5432
  - pgadmin:                Container: naqel_maps_pgadmin, Port: 5050
  - naqel_maps (Django):    Container: Naqel_Maps, Port: 8000

VOLUMES:
  - postgres_data:          PostgreSQL database data
  - pgadmin_data:           PgAdmin configuration and data

================================================================================
                            START SERVICES
================================================================================

# Start all services in background (detached mode)
# WHEN: Starting the project for the first time or after stopping
# WHY:  Runs services in background, returns terminal control immediately
# WHAT: Builds images if needed, creates containers, starts all services
docker-compose up -d

# Start all services and view logs in real-time
# WHEN: Starting services and want to see startup logs immediately
# WHY:  Useful for debugging startup issues or monitoring initialization
# WHAT: Starts services and attaches to log output (Ctrl+C stops services)
docker-compose up

# Start only specific service(s)
# WHEN: You only need database running, not the full application
# WHY:  Saves system resources by running only what you need
# WHAT: Starts only the specified service(s) in detached mode
docker-compose up -d db
docker-compose up -d db pgadmin

================================================================================
                            STOP SERVICES
================================================================================

# Stop all services (keeps containers)
# WHEN: Temporarily stopping without removing containers
# WHY:  Preserves container state, faster to restart later
# WHAT: Stops running containers but keeps them; all data/volumes remain intact
docker-compose stop

# Stop and remove containers (keeps volumes)
# WHEN: Cleaning up containers but want to preserve database data
# WHY:  Removes containers and networks, but keeps volumes for data persistence
# WHAT: Stops containers, removes them, removes networks; volumes remain
docker-compose down

# Stop specific service only
# WHEN: Need to stop only Django app while keeping database running
# WHY:  Useful when you need to restart just one service without affecting others
# WHAT: Stops only the specified service container
docker-compose stop naqel_maps

================================================================================
                           RESTART SERVICES
================================================================================

# Restart all services
# WHEN: Need to restart all services without rebuilding
# WHY:  Quick restart when configuration hasn't changed, just need fresh state
# WHAT: Stops and starts all containers in place (no rebuild)
docker-compose restart

# Restart specific service
# WHEN: One service has issues and needs restart, others are fine
# WHY:  Fixes issues with one service without affecting others
# WHAT: Restarts only the specified container
docker-compose restart db
docker-compose restart naqel_maps

# Clean restart (stop, remove, and start fresh)
# WHEN: Need complete fresh restart with current configuration
# WHY:  Ensures clean state, removes any stale containers/networks
# WHAT: Stops and removes containers/networks, then recreates and starts them
# NOTE: Use --build if you've changed requirements.txt or Dockerfile
docker-compose down && docker-compose up -d
# Or with rebuild: docker-compose down && docker-compose up -d --build

================================================================================
                    DELETE POSTGRES DATA & FRESH START
================================================================================

# Remove all volumes and start fresh (RECOMMENDED - Complete Cleanup)
# WHEN: Want to delete ALL data (PostgreSQL + PgAdmin) and start completely fresh
# WHY:  Complete reset - removes all persistent data including database and PgAdmin
# WHAT: Stops containers, removes them, and deletes ALL volumes (postgres_data + pgadmin_data)
# NOTE: --build flag ensures all dependencies (like Pillow) are properly installed from requirements.txt
docker-compose down -v
docker-compose up -d --build
docker-compose exec naqel_maps python manage.py migrate

# Remove only PostgreSQL volume (keeps PgAdmin data)
# WHEN: Want to reset database but keep PgAdmin settings
# WHY:  Selective cleanup - preserves PgAdmin configuration while resetting database
# WHAT: Stops services, removes only postgres_data volume, then restarts
# NOTE: --build flag ensures all dependencies (like Pillow) are properly installed from requirements.txt
docker-compose down
docker volume rm gis_postgres_data
docker-compose up -d --build
docker-compose exec naqel_maps python manage.py migrate

# Complete cleanup including orphaned resources
# WHEN: Complete cleanup including old/orphaned containers from previous configs
# WHY:  Removes everything including containers from old docker-compose configurations
# WHAT: Full cleanup of containers, networks, volumes, and orphaned resources
# NOTE: --build flag ensures all dependencies (like Pillow) are properly installed from requirements.txt
docker-compose down -v --remove-orphans
docker-compose up -d --build
docker-compose exec naqel_maps python manage.py migrate

# Remove unused volumes (cleanup orphaned volumes)
# WHEN: Want to free disk space from unused volumes
# WHY:  Removes volumes that are not attached to any containers
# WHAT: Deletes all volumes not currently in use by any container
docker volume prune

================================================================================
                        VIEW STATUS & LOGS
================================================================================

# View running containers status
# WHEN: Check which services are currently running
# WHY:  Quick overview of container status, ports, and names
# WHAT: Lists all containers with their status, ports, and container names
docker-compose ps

# View logs from all services
# WHEN: Need to see logs from all services for debugging
# WHY:  Comprehensive view of all service activity
# WHAT: Shows logs from all containers (all services combined)
docker-compose logs

# View logs from specific service
# WHEN: Debugging issues with a specific service
# WHY:  Focused view on one service's logs without noise from others
# WHAT: Shows logs only from the specified service container
docker-compose logs db
docker-compose logs naqel_maps
docker-compose logs pgadmin

# Follow logs in real-time (streaming)
# WHEN: Monitoring services as they run, watching for errors in real-time
# WHY:  See logs as they happen, useful for debugging active issues
# WHAT: Streams logs continuously (press Ctrl+C to exit)
docker-compose logs -f
docker-compose logs -f naqel_maps

# View last N lines of logs
# WHEN: Want to see recent logs without scrolling through old entries
# WHY:  Avoids overwhelming output, shows only recent activity
# WHAT: Displays only the last N lines of logs
docker-compose logs --tail=100 naqel_maps

================================================================================
                        REBUILD & UPDATE
================================================================================

# Rebuild specific service (after code/Dockerfile changes)
# WHEN: Changed Dockerfile, requirements.txt, or dependencies for a service
# WHY:  Rebuilds the image with new changes before restarting
# WHAT: Builds new image for the specified service using updated files
docker-compose build naqel_maps

# Rebuild and restart service in one command
# WHEN: Made changes and want to rebuild and restart immediately
# WHY:  Convenient one-command solution to apply changes and restart
# WHAT: Rebuilds the image and restarts the container in one step
docker-compose up -d --build naqel_maps

# Rebuild all services without cache (complete fresh build)
# WHEN: Suspect caching issues or want guaranteed fresh build
# WHY:  Bypasses Docker cache to ensure all layers are rebuilt from scratch
# WHAT: Rebuilds all images completely without using any cached layers
docker-compose build --no-cache

================================================================================
                        DATABASE ACCESS
================================================================================

# Access PostgreSQL command line (psql)
# WHEN: Need to run SQL queries, inspect database, or manage database directly
# WHY:  Direct database access for advanced operations and debugging
# WHAT: Opens PostgreSQL interactive shell inside the database container
docker-compose exec db psql -U postgres -d naqel_maps

# Alternative: Access using container name
# WHEN: Same as above, alternative method
# WHY:  Direct container access method
# WHAT: Executes psql command directly in the database container
docker exec -it naqel_maps_db psql -U postgres -d naqel_maps

# Run Django migrations
# WHEN: After changing Django models or adding new migrations
# WHY:  Updates database schema to match your Django models
# WHAT: Executes Django migration commands inside the Django container
docker-compose exec naqel_maps python manage.py migrate

# Create Django superuser (admin user)
# WHEN: Setting up admin access to Django admin panel for the first time
# WHY:  Creates admin user account to access Django admin interface
# WHAT: Interactive command to create superuser account (prompts for username, email, password)
docker-compose exec naqel_maps python manage.py createsuperuser

# Run other Django management commands
# WHEN: Need to run any Django management command
# WHY:  Execute Django commands inside the container
# WHAT: Runs specified Django management command
docker-compose exec naqel_maps python manage.py <command>
# Examples:
docker-compose exec naqel_maps python manage.py collectstatic
docker-compose exec naqel_maps python manage.py shell
docker-compose exec naqel_maps python manage.py makemigrations

================================================================================
                        COMMON WORKFLOWS
================================================================================

# Daily Development Workflow
# Start everything for daily work
docker-compose up -d

# Watch logs while developing
docker-compose logs -f naqel_maps

# Stop when done for the day
docker-compose stop

# Complete Fresh Start (Delete All Data)
# Remove everything including all data
docker-compose down -v

# Start fresh with rebuild to ensure all dependencies are installed (IMPORTANT: Prevents Pillow errors)
docker-compose up -d --build

# Run migrations on fresh database (REQUIRED after fresh start)
docker-compose exec naqel_maps python manage.py migrate

# (Optional) Create superuser
docker-compose exec naqel_maps python manage.py createsuperuser

# After Code Changes
# Rebuild and restart application
docker-compose build naqel_maps
docker-compose up -d naqel_maps

# Or in one command
docker-compose up -d --build naqel_maps

================================================================================
                        TROUBLESHOOTING
================================================================================

# Check if containers are running
docker-compose ps

# Check container health
docker-compose ps --format json

# View detailed logs for debugging
docker-compose logs --tail=200 naqel_maps

# Restart a stuck service
docker-compose restart naqel_maps

# Force remove and recreate a service
docker-compose up -d --force-recreate naqel_maps

# Check volume usage
docker volume ls
docker volume inspect gis_postgres_data

# Access container shell for debugging
docker-compose exec naqel_maps sh
docker-compose exec db sh

# Fix "Pillow is not installed" or other dependency errors
# WHEN: Getting errors about missing packages after fresh start
# WHY:  Container wasn't rebuilt, so dependencies from requirements.txt weren't installed
# WHAT: Rebuilds container to install all dependencies properly
docker-compose up -d --build naqel_maps
docker-compose exec naqel_maps python manage.py migrate

================================================================================
                        NOTES
================================================================================

- IMPORTANT: Always use --build flag when doing fresh start to ensure dependencies are installed
  This prevents errors like "Pillow is not installed" by rebuilding the container with requirements.txt

- Always run migrations after fresh database start: 
  docker-compose exec naqel_maps python manage.py migrate

- PostgreSQL data persists in volume: gis_postgres_data
- PgAdmin data persists in volume: gis_pgadmin_data

- To completely reset everything (RECOMMENDED WORKFLOW):
  1. docker-compose down -v
  2. docker-compose up -d --build  (--build ensures Pillow and all dependencies are installed)
  3. docker-compose exec naqel_maps python manage.py migrate
  4. docker-compose exec naqel_maps python manage.py createsuperuser

- Access PgAdmin at: http://localhost:5050
- Access Django app at: http://localhost:8000
- PostgreSQL accessible at: localhost:5432
